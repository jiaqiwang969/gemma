  ---
  组件 1: 骨干网络 (Backbone) - 就是 Gemma-3n

  # modeling_colgemma3n.py 第 76-82 行
  self.model = Gemma3nModel(config)  # ← 直接用 Gemma-3n！

  作用：将图像/文本转换为语义表示

  输入: 文档图像 (800x1000 像素)
        ↓
     MobileNetV5 视觉编码器
        ↓
     256 个视觉 token
        ↓
     Gemma-3n Transformer
        ↓
  输出: hidden_states (1, 256, 2048)
        每个 token 是一个 2048 维向量

  ---
  组件 2: 投影层 (Projection) - 一个线性层

  # modeling_colgemma3n.py 第 96 行
  self.custom_text_proj = nn.Linear(hidden_size, self.dim)  # 2048 → 128

  作用：降维 + 归一化

  # modeling_colgemma3n.py 第 222-225 行
  proj = self.custom_text_proj(hidden_states)  # (batch, seq_len, 128)
  proj = proj / (proj.norm(dim=-1, keepdim=True) + 1e-8)  # L2 归一化

  为什么要降维？
  - 2048 维太大，存储和计算成本高
  - 128 维足够保留语义信息
  - L2 归一化让向量长度为 1，方便计算余弦相似度

  ---
  组件 3: MaxSim 评分器 - 核心算法

  # processing_utils.py 第 179 行
  scores = torch.einsum("bnd,csd->bcns", qs_batch, ps_batch).max(dim=3)[0].sum(dim=2)

  这行代码做了什么？

  Query:    "What is the revenue?"
            ↓ 编码后
            [q1, q2, q3, q4, q5]  ← 5 个 128 维向量

  Document: 财务报表图像
            ↓ 编码后
            [d1, d2, d3, ..., d256]  ← 256 个 128 维向量

  MaxSim 计算过程：

  对于每个 query token qi:
      找到与所有 doc tokens 最相似的那个
      max_score_i = max(qi · d1, qi · d2, ..., qi · d256)

  最终分数 = sum(max_score_1, max_score_2, ..., max_score_5)

  直观理解：
  - q1 = "What" → 可能匹配 d100 (某个通用词)
  - q2 = "revenue" → 匹配 d50 (文档中 "$52.8M" 附近的 token)
  - 每个 query 词都找到文档中最相关的位置
  - 总分 = 所有匹配分数之和

  训练：对比学习损失

  # late_interaction_losses.py 第 140-164 行
  class ColbertLoss:
      def forward(self, query_embeddings, doc_embeddings):
          # 1. 计算所有 query-doc 对的 MaxSim 分数
          raw = torch.einsum("bnd,csd->bcns", query_embeddings, doc_embeddings)
          scores = raw.max(dim=3)[0].sum(dim=2)  # MaxSim

          # 2. 对角线是正样本 (query_i 对应 doc_i)
          # 3. 用 CrossEntropy 让正样本分数最高
          return self.ce_loss(scores / self.temperature, pos_idx)

  训练目标：让 query 和对应的正确文档分数最高

  def generate_retrieval_samples(logs):
      samples = []
      for log in logs:
          if log["retrieval"]["retrieval_feedback"] == "correct":
              # 用户确认检索正确 → 正样本
              samples.append({
                  "query": log["query"]["text"],
                  "positive_doc": log["retrieval"]["selected_doc"],
                  "negative_docs": [c["doc_id"] for c in log["retrieval"]["candidates"]
                                    if c["doc_id"] != log["retrieval"]["selected_doc"]]
              })
          elif log["retrieval"]["retrieval_feedback"] == "wrong":
              # 用户说检索错误 → 学习什么是错的
              # 可以用于 hard negative mining
              pass
      return samples

  2. 生成训练数据 (SFT)

  # 从日志生成生成训练样本
  def generate_sft_samples(logs):
      samples = []
      for log in logs:
          if log["generation"]["response_feedback"] == "positive":
              # 用户点赞 → 好的回答
              samples.append({
                  "instruction": log["query"]["text"],
                  "context_image": log["generation"]["context_doc"],
                  "response": log["generation"]["response"]
              })
      return samples

  ---
  隐式反馈信号

  VR 场景下，用户反馈不再需要显式点赞/踩，而是通过行为推断：

  | 信号     | 正反馈             | 负反馈            |
  |----------|--------------------|-------------------|
  | 注视时间 | 长时间注视 AI 回答 | 快速移开视线      |
  | 后续行为 | 继续当前任务       | 重复提问/换个问法 |
  | 语音情绪 | 平静/满意语调      | 沮丧/疑惑语调     |
  | 头部动作 | 点头               | 摇头              |
  | 手势     | 无                 | 挥手取消          |

  def infer_feedback(fragment):
      """从隐式信号推断用户反馈"""
      score = 0.5  # 中性起点

      # 注视时间
      if fragment.attention.dwell_time > 5.0:
          score += 0.2  # 长时间注视 = 可能有用
      elif fragment.attention.dwell_time < 1.0:
          score -= 0.2  # 快速移开 = 可能没用

      # 后续行为
      if fragment.interaction.follow_up is None:
          score += 0.1  # 没有追问 = 可能满意
      elif "再说一遍" in fragment.interaction.follow_up:
          score -= 0.3  # 要求重复 = 没听懂/不满意

      # 语音情绪
      if fragment.audio.emotion == "satisfied":
          score += 0.2
      elif fragment.audio.emotion == "frustrated":
          score -= 0.3

      return score  # 0-1 之间的满意度分数

  # projection_head.py
  class ProjectionHead(nn.Module):
      """投影层: 将 Gemma-3n hidden_states 投影到检索空间"""

      def __init__(self, hidden_size: int = 2048, embedding_dim: int = 128):
          super().__init__()
          self.proj = nn.Linear(hidden_size, embedding_dim)
          self._init_weights()

      def forward(self, hidden_states: torch.Tensor) -> torch.Tensor:
          # hidden_states: (batch, seq_len, hidden_size)
          proj = self.proj(hidden_states)  # (batch, seq_len, embedding_dim)
          proj = F.normalize(proj, p=2, dim=-1)  # L2 归一化
          return proj
